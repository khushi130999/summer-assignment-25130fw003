2. write a program to print multiplication table of a number ..

#include <iostream>
using namespace std;

int main() 
{
  int n;
  cout<<"enter number :";
  cin>> n;
  cout<< n<< "\n";
  for(int i=1;i<=10;i++){
    cout<< n*i<< endl;
  }
    return 0;
}
