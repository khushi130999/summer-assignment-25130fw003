11. find the GCD of two numbers..
#include <iostream>
using namespace std;

int main() 
{
  
    int a , b;
    cout<< "enter two numbers";
    cin>>a>>b;
    while (b!=0){
      int temp = b;
      b= a%b;
      a= temp;
    }
    cout << " gcd ="<< a;
    return 0;
}

12. find lcm of two numbers ..

#include <iostream>
using namespace std;

int main() 
{
  
    int a , b, x ,y;
    cout<< "enter two numbers";
    cin>>a>>b;
    
    x=a;
    y=b;
    
    while (y!=0){
      int temp = y;
      y= x%y;
      x= temp;
    }
    int gcd = x;
    int lcm = (a*b)/gcd;
    
    cout << " lcm ="<< lcm;
    return 0;
}
